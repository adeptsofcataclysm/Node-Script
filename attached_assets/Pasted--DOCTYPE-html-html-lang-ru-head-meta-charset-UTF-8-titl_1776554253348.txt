<script>
    let bulletPos = -1; 
    let currentPos = 0; 
    let rotation = 0;
    let currentPlayer = 1;
    let scores = { 1: 0, 2: 0 };
    let isGameOver = false;

    const sounds = {
        spin: new Audio('https://soundjay.com'),
        click: new Audio('https://soundjay.com'),
        bang: new Audio('https://soundjay.com')
    };

    function spin() {
        if (isGameOver) return;
        document.getElementById('spinBtn').disabled = true;
        bulletPos = Math.floor(Math.random() * 6);
        currentPos = 0;
        rotation += 1440 + (Math.floor(Math.random() * 6) * 60);
        document.getElementById('drum').style.transform = `rotate(${rotation}deg)`;
        sounds.spin.play();
        setTimeout(() => {
            document.getElementById('shootBtn').disabled = false;
            document.getElementById('spinBtn').disabled = false;
        }, 2000);
    }

    function shoot() {
        if (isGameOver || bulletPos === -1) return;

        if (currentPos === bulletPos) {
            sounds.bang.play();
            isGameOver = true;
            document.getElementById(`ch${currentPos}`).classList.add('loaded');
            const winner = currentPlayer === 1 ? 2 : 1;
            document.getElementById('win-msg').innerHTML = `ИГРОК ${currentPlayer} ВЫБЫЛ<br><span style="color:var(--accent)">ПОБЕДА ИГРОКА ${winner}</span>`;
            document.getElementById('wasted').style.display = 'flex';
        } else {
            sounds.click.play();
            scores[currentPlayer]++;
            updateUI();
            
            // Смена игрока
            currentPlayer = currentPlayer === 1 ? 2 : 1;
            document.getElementById('turn-announcer').innerText = `Ход Игрока ${currentPlayer}`;
            document.getElementById('p1-ui').classList.toggle('active');
            document.getElementById('p2-ui').classList.toggle('active');

            currentPos = (currentPos + 1) % 6;
            rotation += 60;
            document.getElementById('drum').style.transform = `rotate(${rotation}deg)`;
        }
    }

    function updateUI() {
        document.getElementById('p1-score').innerText = scores[1];
        document.getElementById('p2-score').innerText = scores[2];
    }

    function restart() {
        scores = { 1: 0, 2: 0 };
        currentPlayer = 1;
        isGameOver = false;
        bulletPos = -1;
        updateUI();
        document.getElementById('wasted').style.display = 'none';
        document.getElementById('p1-ui').classList.add('active');
        document.getElementById('p2-ui').classList.remove('active');
        document.getElementById('turn-announcer').innerText = `Ход Игрока 1`;
        document.getElementById('shootBtn').disabled = true;
        document.querySelectorAll('.chamber').forEach(c => c.classList.remove('loaded'));
    }
</script>
